<script>
  // Service workers require http/https. Opening index.html straight from disk
  // (file://) is perfectly fine for previewing, but registration is impossible
  // there - so don't attempt it, and don't warn about it either.
  (function () {
    var served = location.protocol === "http:" || location.protocol === "https:";
    if (!("serviceWorker" in navigator) || !served) return;
    window.addEventListener("load", function () {
      navigator.serviceWorker.register("sw.js").catch(function (err) {
        // Don't assume a cause. Registration is also blocked by policy in
        // in-app browsers (iOS WKWebView) and some private modes - neither of
        // which the deployment can do anything about. So check whether the
        // supporting files are genuinely reachable before blaming them.
        var detail = (err && err.message) ? err.message : String(err);
        fetch("sw.js", { method: "HEAD" })
          .then(function (r) {
            if (!r.ok) {
              console.warn(
                "Offline support unavailable: sw.js returned " + r.status +
                ". Check that sw.js sits in the same folder as index.html."
              );
            } else {
              console.info(
                "Offline support unavailable in this browser (" + detail + "). " +
                "Supporting files are present, so this is a browser restriction - " +
                "commonly an in-app browser. The assessment tool itself is unaffected."
              );
            }
          })
          .catch(function () {
            console.info("Offline support unavailable. The assessment tool itself is unaffected.");
          });
      });
    });
  })();
</script>
